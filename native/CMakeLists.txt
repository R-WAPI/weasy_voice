cmake_minimum_required(VERSION 3.22)
project(weasy_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../whisper-src" CACHE PATH "Pinned whisper.cpp source")
add_subdirectory("${WHISPER_SOURCE}" whisper-build EXCLUDE_FROM_ALL)
add_library(weasy_whisper SHARED weasy_whisper.cpp)
target_link_libraries(weasy_whisper PRIVATE whisper android log)
target_link_options(weasy_whisper PRIVATE "-Wl,-z,max-page-size=16384")
